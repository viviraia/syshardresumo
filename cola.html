Modelo: fork + execvp + waitpid (com retry)

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>

int main(int argc, char *argv[]) {
    if (argc != 3) { printf("USO: %s <a> <b>\n", argv[0]); return 1; }
    int ret = 1;
    while (ret > 0) {
        pid_t filho = fork();
        if (filho == 0) {
            char *args[] = {"./programa", argv[1], argv[2], NULL};
            execvp(args[0], args);
            perror("execvp");
            exit(1);
        }
        int wst;
        waitpid(filho, &wst, 0);
        if (WIFEXITED(wst)) {
            ret = (char) WEXITSTATUS(wst);      // (char) recupera retornos negativos
            printf("retornou %d\n", ret);
        } else if (WIFSIGNALED(wst)) {
            printf("morreu com sinal %d (%s)\n", WTERMSIG(wst), strsignal(WTERMSIG(wst)));
            ret = 1;                            // tenta de novo
        }
    }
    return 0;
}

Modelo: handler que salva em arquivo e termina pelo sinal

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <fcntl.h>

long estado = 0;                                  // global!

void salva(void) {
    int fd = open("estado.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) { perror("open"); return; }
    char buf[64];
    int n = snprintf(buf, sizeof(buf), "%ld\n", estado);
    write(fd, buf, n);
    close(fd);
}

void handler(int sig) {
    salva();
    printf("salvo: %ld\n", estado);
    struct sigaction d;                           // termina "de verdade" por SIGINT
    d.sa_handler = SIG_DFL;
    d.sa_flags = 0;
    sigemptyset(&d.sa_mask);
    sigaction(SIGINT, &d, NULL);
    raise(SIGINT);
}

int main(void) {
    printf("PID: %d\n", getpid());
    struct sigaction s;
    s.sa_handler = handler;
    s.sa_flags = 0;
    sigemptyset(&s.sa_mask);
    sigaddset(&s.sa_mask, SIGTERM);               // opcional: bloqueia SIGTERM no handler
    sigaction(SIGINT, &s, NULL);
    while (1) { estado++; usleep(1000); }
}

Modelo: read_line com read (tamanho arbitrário)

char *read_line(int fd) {
    int cap = 16, len = 0, lidos;
    char *buf = malloc(cap);
    char c;
    while ((lidos = read(fd, &c, 1)) > 0) {
        if (c == '\n') break;
        if (len + 1 >= cap) { cap *= 2; buf = realloc(buf, cap); }
        buf[len++] = c;
    }
    if (lidos <= 0 && len == 0) { free(buf); return NULL; }
    buf[len] = '\0';
    return buf;
}

Modelo: N threads, struct por thread, limite por semáforo e mutex

#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <semaphore.h>

typedef struct {
    int id;
    int entrada;
    int res;                     // saída
    sem_t *limite;
    pthread_mutex_t *mutex;
} t_arg;

long total = 0;

void *tarefa(void *_arg) {
    t_arg *a = _arg;
    sem_wait(a->limite);         // entra (no máx. K ao mesmo tempo)
    a->res = a->entrada * 2;     // trabalho
    sem_post(a->limite);         // sai
    pthread_mutex_lock(a->mutex);
    total += a->res;             // região crítica
    pthread_mutex_unlock(a->mutex);
    return NULL;
}

int main(int argc, char *argv[]) {
    int n = 10, k = (argc > 1) ? atoi(argv[1]) : 2;
    sem_t limite;
    sem_init(&limite, 0, k);
    pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
    pthread_t *tids = malloc(sizeof(pthread_t) * n);
    t_arg **args = malloc(sizeof(t_arg *) * n);
    for (int i = 0; i < n; i++) {
        args[i] = malloc(sizeof(t_arg));
        args[i]->id = i;
        args[i]->entrada = i;
        args[i]->limite = &limite;
        args[i]->mutex = &mutex;
        pthread_create(&tids[i], NULL, tarefa, args[i]);
    }
    for (int i = 0; i < n; i++) {
        pthread_join(tids[i], NULL);
        printf("THREAD %02d RES %d\n", args[i]->id, args[i]->res);
        free(args[i]);
    }
    printf("TOTAL: %ld\n", total);
    free(args);
    free(tids);
    sem_destroy(&limite);
    return 0;
}

Modelo: diagrama de dependências

#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

sem_t sA, sB;                                     // um por parte que tem setas saindo (init 0)

void *t1(void *a) { printf("A\n"); sem_post(&sA); sem_post(&sA); return NULL; }  // A -> C e A -> D
void *t2(void *a) { printf("B\n"); sem_post(&sB); return NULL; }                 // B -> C
void *t3(void *a) { sem_wait(&sA); sem_wait(&sB); printf("C\n"); return NULL; }  // C depende de A e B
void *t4(void *a) { sem_wait(&sA); printf("D\n"); return NULL; }                 // D depende de A

int main(void) {
    sem_init(&sA, 0, 0);
    sem_init(&sB, 0, 0);
    void *(*f[4])(void *) = {t1, t2, t3, t4};
    pthread_t tid[4];
    for (int i = 0; i < 4; i++) pthread_create(&tid[i], NULL, f[i], NULL);
    for (int i = 0; i < 4; i++) pthread_join(tid[i], NULL);
    sem_destroy(&sA);
    sem_destroy(&sB);
    return 0;
}

Modelo: barreira para N threads

#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#define N 4

int chegaram = 0;
pthread_mutex_t m = PTHREAD_MUTEX_INITIALIZER;
sem_t barreira;                                   // init 0

void *tarefa(void *a) {
    printf("Antes da barreira\n");
    pthread_mutex_lock(&m);
    chegaram++;
    int ultimo = (chegaram == N);
    pthread_mutex_unlock(&m);
    if (ultimo) { for (int i = 0; i < N - 1; i++) sem_post(&barreira); }
    else sem_wait(&barreira);
    printf("Depois da barreira\n");
    return NULL;
}

int main(void) {
    sem_init(&barreira, 0, 0);
    pthread_t t[N];
    for (int i = 0; i < N; i++) pthread_create(&t[i], NULL, tarefa, NULL);
    for (int i = 0; i < N; i++) pthread_join(t[i], NULL);
    sem_destroy(&barreira);
    return 0;
}

Erros que mais tiram ponto

Processos
  • Esquecer o NULL no fim do vetor do execvp, ou esquecer o argv[0].
  • Código depois do execvp sem exit: se o exec falhar, o filho continua o laço e cria mais processos.
  • Ler WEXITSTATUS sem cast (char), de modo que -1 vira 255.
  • Usar system() quando é proibido.
  • Não esperar os filhos (deixa zumbis e o pai sai antes).
Arquivos
  • Usar fopen/fprintf/fgets/getline quando o enunciado pede só open/read/write.
  • Esquecer o O_CREAT ou o mode (0644), ou o O_TRUNC (o arquivo fica com restos do conteúdo anterior).
  • Escrever sizeof(buf) bytes em vez de n, o que grava lixo.
  • Esquecer o '\0' depois do read antes de usar como string.
Sinais
  • Registrar o handler depois do laço infinito: ele nunca é registrado.
  • Assinatura do handler diferente de void h(int).
  • Esquecer o sigemptyset(&s.sa_mask) e o s.sa_flags = 0 (lixo na struct).
  • O dado que o handler precisa não está numa global.
Threads e semáforos
  • Compilar sem -pthread.
  • Passar &i do laço ou o endereço de uma local de função que já retornou.
  • Não fazer join em todas: o main termina e mata as threads.
  • Ler o resultado antes do join.
  • free antes do join, ou não liberar nada (o valgrind acusa).
  • Usar join ou sleep para impor ordem (cria dependências extras, e a questão vale no máximo 50%).
  • Semáforo de dependência iniciado em 1, ou poucos post para muitos dependentes, levando a deadlock.
  • sem_wait com o mutex travado, levando a deadlock.
  • Mudar ou adicionar printf quando o enunciado proíbe, ou imprimir fora do formato exato (espaços e \n extras).

Estratégia para as 3 horas

  1. Leia o README inteiro e anote as rubricas em %: elas dizem o que priorizar.
  2. Comece pela questão que você domina mais. Mire primeiro nos itens de maior peso.
  3. Entregue a cada questão terminada: o Blackboard aceita várias tentativas. Rezipe a pasta inteira e não renomeie os arquivos.
  4. Compile sempre com -Wall e teste com os programas de referência (q2_ref) quando houver.
  5. No fim, rode o valgrind e confira o formato exato da saída.
