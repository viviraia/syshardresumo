Questão 1 — Semáforos 2,5 pontos

O arquivo q1.c tem 5 funções de thread. As partes de cada uma (os printf de uma letra) têm as relações de dependência da figura: uma seta X → Y significa que Y só pode ser impresso depois de X. As partes de uma mesma thread executam na ordem do código.

Em texto: A → B, A → D, B → E, C → E, D → F, E → G, F → G. Pela ordem do código, A vem antes de E (thread1) e C antes de F (thread3).

Seu trabalho:

  1. criar as 5 threads em um laço para execução concorrente e esperar todas (40%);
  2. usar semáforos para que a ordem dos prints respeite a figura (60%).

OBS:

  • Não introduza novas dependências: se o programa impuser relações além das da figura, a questão vale no máximo 50%.
  • Não altere os printf existentes nem adicione novos.
  • Compile com gcc -Wall -g q1.c -o q1 -pthread.
⬇ q1.c (base)
Ver gabarito comentado da Q1

Ideia: um semáforo iniciado em 0 para cada parte que tem setas saindo (A, B, C, D, E, F). Depois de imprimir X, faça um sem_post(sX) por seta que sai de X; antes de imprimir Y, faça um sem_wait em cada semáforo de que Y depende. A sai com duas setas, então recebe dois posts (B e D consomem um cada).

Ordens válidas (de 300 execuções do gabarito): ACBDEFG, CADBFEG, ADCFBEG… Inválidas: BACDEFG (B antes de A) e ACBDEGF (G antes de F).

// GABARITO - Prova Modelo SisHard - Questao 1
// Compile: gcc -Wall -g q1.c -o q1 -pthread
//
// Dependencias do diagrama:
//   A -> B     A -> D
//   B -> E     C -> E
//   D -> F
//   E -> G     F -> G
//
// Regra: um semaforo (iniciado em 0) por "aresta de saida" de cada parte.
// Quem termina a parte X faz sem_post(sX) uma vez para CADA parte que depende dela.
// Quem depende de X faz sem_wait(sX) antes de executar.
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <semaphore.h>

typedef struct {
    sem_t sA, sB, sC, sD, sE, sF;
} t_arg;

void *thread1(void *_arg) {
    t_arg *arg = _arg;
    printf("A\n");
    sem_post(&arg->sA);   // libera B
    sem_post(&arg->sA);   // libera D

    sem_wait(&arg->sB);   // E depende de B
    sem_wait(&arg->sC);   // E depende de C
    printf("E\n");
    sem_post(&arg->sE);   // libera G
    return NULL;
}

void *thread2(void *_arg) {
    t_arg *arg = _arg;
    sem_wait(&arg->sA);   // B depende de A
    printf("B\n");
    sem_post(&arg->sB);
    return NULL;
}

void *thread3(void *_arg) {
    t_arg *arg = _arg;
    printf("C\n");        // C nao depende de ninguem
    sem_post(&arg->sC);

    sem_wait(&arg->sD);   // F depende de D
    printf("F\n");
    sem_post(&arg->sF);
    return NULL;
}

void *thread4(void *_arg) {
    t_arg *arg = _arg;
    sem_wait(&arg->sA);   // D depende de A
    printf("D\n");
    sem_post(&arg->sD);
    return NULL;
}

void *thread5(void *_arg) {
    t_arg *arg = _arg;
    sem_wait(&arg->sE);   // G depende de E
    sem_wait(&arg->sF);   // G depende de F
    printf("G\n");
    return NULL;
}

int main(int argc, char *argv[]) {
    void *(*funcs[5])(void *) = {thread1, thread2, thread3, thread4, thread5};
    pthread_t tids[5];

    t_arg *arg = malloc(sizeof(t_arg));
    sem_init(&arg->sA, 0, 0);
    sem_init(&arg->sB, 0, 0);
    sem_init(&arg->sC, 0, 0);
    sem_init(&arg->sD, 0, 0);
    sem_init(&arg->sE, 0, 0);
    sem_init(&arg->sF, 0, 0);

    for (int i = 0; i < 5; i++) {
        pthread_create(&tids[i], NULL, funcs[i], arg);
    }
    for (int i = 0; i < 5; i++) {
        pthread_join(tids[i], NULL);
    }

    sem_destroy(&arg->sA);
    sem_destroy(&arg->sB);
    sem_destroy(&arg->sC);
    sem_destroy(&arg->sD);
    sem_destroy(&arg->sE);
    sem_destroy(&arg->sF);
    free(arg);
    return 0;
}

Como testar: for i in $(seq 50); do ./q1 | tr -d '\n'; echo; done | sort | uniq -c. Várias ordens diferentes devem aparecer, todas válidas. Se sair sempre a mesma ordem, desconfie de dependência extra.

Questão 2 — Processos 2,5 pontos

O programa instavel (fonte em instavel.c; compile-o, mas não o altere) processa uma tarefa, mas é instável. Ele é chamado como ./instavel <tarefa> <n> e:

  • retorna -1 se a quantidade de argumentos estiver errada ou se n for negativo;
  • retorna 0 quando a tarefa é processada com sucesso;
  • retorna 1, 2 ou 3 em falha temporária;
  • às vezes trava e morre com um sinal (SIGABRT).

Escreva q2 (arquivo q2.c), chamado como ./q2 <tarefa> <n>, que executa ./instavel repetidamente, cada vez num processo filho, até que ele dê certo ou indique erro de parâmetro. A cada tentativa, imprima exatamente:

Tentativa <k>: instavel retornou <ret>                 (terminou normalmente)
Tentativa <k>: instavel terminou com o sinal <s> (<nome>) (morreu por sinal)
SUCESSO apos <k> tentativa(s)                           (quando ret == 0)
ERRO: parametros invalidos, desistindo                  (quando ret < 0)

Rubrica:

  • 30%: fork + execvp com os argumentos corretos (o filho repassa argv[1] e argv[2]);
  • 40%: o pai espera, lê o retorno corretamente (inclusive o -1), repete nas falhas 1–3 e para em 0 ou em -1;
  • 20%: detecta o término por sinal e mostra o número e o nome (strsignal), tentando de novo;
  • 10%: formato exato da saída e contagem das tentativas.

Não pode usar system. Compile com gcc -Wall -g q2.c -o q2. Teste com ./q2 tarefa 7 e com ./q2 tarefa -3.

Pontos-chave:

  • o vetor do exec começa com o nome do programa e termina com NULL;
  • o perror + exit depois do execvp evita que o filho continue o laço se o exec falhar;
  • (char) WEXITSTATUS transforma o 255 em -1;
  • WIFSIGNALED trata o abort().
// GABARITO - Prova Modelo SisHard - Questao 2
// Compile: gcc -Wall -g q2.c -o q2
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/types.h>
#include <sys/wait.h>

int main(int argc, char *argv[]) {
    if (argc != 3) {
        printf("USO: ./q2 <tarefa> <n>\n");
        return 1;
    }

    int tentativa = 0;
    int continuar = 1;

    while (continuar) {
        tentativa++;

        pid_t filho = fork();
        if (filho == 0) {
            // FILHO: vira o programa instavel.
            // argv sempre comeca com o nome do programa e termina com NULL
            char *args[] = {"./instavel", argv[1], argv[2], NULL};
            execvp(args[0], args);
            // so chega aqui se o exec FALHAR
            perror("execvp");
            exit(127);
        }

        // PAI
        int wst;
        waitpid(filho, &wst, 0);

        if (WIFEXITED(wst)) {
            // WEXITSTATUS devolve so 8 bits (0..255). "return -1" vira 255.
            // O cast para (char) recupera o valor com sinal (-1).
            int ret = (char) WEXITSTATUS(wst);
            printf("Tentativa %d: instavel retornou %d\n", tentativa, ret);

            if (ret == 0) {
                printf("SUCESSO apos %d tentativa(s)\n", tentativa);
                continuar = 0;
            } else if (ret < 0) {
                printf("ERRO: parametros invalidos, desistindo\n");
                continuar = 0;
            }
            // ret entre 1 e 3: falha temporaria -> tenta de novo
        } else if (WIFSIGNALED(wst)) {
            int sig = WTERMSIG(wst);
            printf("Tentativa %d: instavel terminou com o sinal %d (%s)\n",
                   tentativa, sig, strsignal(sig));
            // terminou de forma anormal -> tenta de novo
        }
    }

    return 0;
}

Questão 3 — Sinais e arquivos 2,5 pontos

Em q3.c, a função trabalha() incrementa para sempre a variável global contador. Queremos poder interromper e retomar esse trabalho. Altere o programa (sem mexer em trabalha()) para que:

  1. mostre o PID do processo no início (10%);
  2. ao receber SIGUSR1, imprima Progresso: <contador> e continue rodando (15%);
  3. ao receber SIGINT, grave o valor atual de contador em formato texto no arquivo estado.txt (sobrescrevendo o anterior) e imprima Estado salvo: <contador> (35%);
  4. depois de salvar, o processo deve terminar por causa do SIGINT, de modo que echo $? mostre 130 (10%);
  5. ao iniciar, se estado.txt existir, leia o valor salvo e continue a contagem a partir dele (20%);
  6. enquanto o handler de SIGINT roda, os sinais SIGTERM e SIGUSR1 devem ficar bloqueados (10%).

OBS:

  • Os handlers devem ser funções próprias, registradas na main com sigaction.
  • Para arquivos, use apenas open, read, write e close (nada de fopen/fprintf/fscanf).
  • Teste com ./q3 &, kill -USR1 <pid>, kill -INT <pid> e cat estado.txt. Depois rode de novo e confira que a contagem continuou.

Pontos-chave:

  • o número vira texto com snprintf, e o write usa o tamanho que ele devolve;
  • O_TRUNC sobrescreve o arquivo;
  • sigaddset na sa_mask do SIGINT faz o bloqueio;
  • para terminar pelo sinal, volte ao SIG_DFL e chame raise(SIGINT);
  • a leitura do estado usa read até o EOF e depois atol.
// GABARITO - Prova Modelo SisHard - Questao 3
// Compile: gcc -Wall -g q3.c -o q3
// Teste:   ./q3 &     kill -USR1 <pid>     kill -INT <pid>     cat estado.txt
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <signal.h>
#include <fcntl.h>
#include <sys/types.h>
#include <sys/stat.h>

long contador = 0;

void trabalha() {
    while (1) {
        contador++;
        if (contador % 50 == 0) {
            usleep(1000);
        }
    }
}

// Salva o contador em formato TEXTO usando so open/write/close
void salva_estado() {
    int fd = open("estado.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) {
        perror("open");
        return;
    }
    char buf[64];
    int n = snprintf(buf, sizeof(buf), "%ld\n", contador); // numero -> texto
    write(fd, buf, n);
    close(fd);
}

// Le estado.txt (se existir). Retorna 0 se o arquivo nao existir.
long le_estado() {
    int fd = open("estado.txt", O_RDONLY);
    if (fd == -1) {
        return 0; // ainda nao existe: comeca do zero
    }
    char buf[64];
    int total = 0, lidos;
    // read pode devolver menos bytes que o pedido: le ate o EOF (retorno 0)
    while (total < (int)sizeof(buf) - 1 &&
           (lidos = read(fd, buf + total, sizeof(buf) - 1 - total)) > 0) {
        total += lidos;
    }
    buf[total] = '\0';
    close(fd);
    return atol(buf); // texto -> numero
}

void handler_usr1(int sig) {
    printf("Progresso: %ld\n", contador);
}

void handler_int(int sig) {
    salva_estado();
    printf("\nEstado salvo: %ld\n", contador);

    // Volta ao comportamento padrao e reenvia o sinal:
    // assim o processo termina DE VERDADE por SIGINT (o shell fica sabendo).
    struct sigaction padrao;
    padrao.sa_handler = SIG_DFL;
    padrao.sa_flags = 0;
    sigemptyset(&padrao.sa_mask);
    sigaction(SIGINT, &padrao, NULL);
    raise(SIGINT); // ou kill(getpid(), SIGINT);
}

int main() {
    printf("Meu PID: %d\n", getpid());

    contador = le_estado();
    printf("Comecando de %ld\n", contador);

    // SIGINT: durante o handler, bloqueia SIGTERM e SIGUSR1
    struct sigaction s_int;
    s_int.sa_handler = handler_int;
    s_int.sa_flags = 0;
    sigemptyset(&s_int.sa_mask);
    sigaddset(&s_int.sa_mask, SIGTERM);
    sigaddset(&s_int.sa_mask, SIGUSR1);
    sigaction(SIGINT, &s_int, NULL);

    // SIGUSR1: imprime o progresso
    struct sigaction s_usr1;
    s_usr1.sa_handler = handler_usr1;
    s_usr1.sa_flags = 0;
    sigemptyset(&s_usr1.sa_mask);
    sigaction(SIGUSR1, &s_usr1, NULL);

    trabalha();
    return 0;
}

Questão 4 — Threads e E/S 2,5 pontos

O programa q4 é chamado como ./q4 <max_threads> <arquivo>. Cada linha do arquivo tem números inteiros separados por espaços (as linhas podem ser muito longas). Para cada linha, crie uma thread que chama soma_thread. Essa thread soma os números com soma_linha (que demora 1 s), guarda o resultado na struct e acumula em total_geral.

Suas tarefas:

  • implementar char *read_line(int fd) usando apenas read, aceitando linhas de tamanho arbitrário; com tamanho máximo fixo, metade desta nota (30%);
  • completar a main: uma thread por linha (o argumento de cada uma no heap), esperar todas e imprimir, na ordem das linhas, LINHA %02d SOMA %ld e, no fim, TOTAL: %ld (20%);
  • fazer as threads respeitarem o limite max_threads de execuções simultâneas de soma_linha: todas são criadas, mas esperam a vez (20%);
  • eliminar a race condition em total_geral (10%);
  • o programa passa no valgrind --leak-check=full sem erros nem vazamentos (20%, só se o resto estiver correto).

Com o numeros.txt fornecido (6 linhas, sendo a 4ª uma linha longa e a 5ª com números negativos), a saída esperada é:

LINHA 00 SOMA 10
LINHA 01 SOMA 60
LINHA 02 SOMA 100
LINHA 03 SOMA 500
LINHA 04 SOMA -3
LINHA 05 SOMA 42
TOTAL: 709

Com time ./q4 2 numeros.txt o tempo deve ser de cerca de 3 s (6 linhas, 2 por vez); com max_threads = 6, cerca de 1 s. Imprima exatamente o formato acima, sem espaços extras. Compile com gcc -Wall -g q4.c -o q4 -pthread.

Pontos-chave:

  • o read_line dobra o buffer com realloc e trata a última linha sem \n;
  • o semáforo limite começa com max_threads e envolve só o trabalho;
  • o mutex protege total_geral;
  • os resultados são impressos depois do join de cada thread, e cada free vem depois do respectivo join.
// GABARITO - Prova Modelo SisHard - Questao 4
// Compile: gcc -Wall -g q4.c -o q4 -pthread
// Teste:   time ./q4 2 numeros.txt     (6 linhas, 2 por vez -> ~3 s)
//          valgrind --leak-check=full ./q4 3 numeros.txt
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <pthread.h>
#include <semaphore.h>

typedef struct {
    int id;
    char *linha;
    long soma;
    sem_t *limite;            // limita quantas threads trabalham ao mesmo tempo
    pthread_mutex_t *mutex;   // protege total_geral
} t_arg;

long total_geral = 0;

long soma_linha(char *linha) {
    sleep(1);
    long soma = 0;
    char *p = linha;
    char *fim;
    while (1) {
        long v = strtol(p, &fim, 10);
        if (p == fim) break;
        soma += v;
        p = fim;
    }
    return soma;
}

void *soma_thread(void *_arg) {
    t_arg *arg = _arg;

    sem_wait(arg->limite);          // pede uma "vaga" (bloqueia se ja ha max_threads)
    arg->soma = soma_linha(arg->linha);
    sem_post(arg->limite);          // devolve a vaga

    pthread_mutex_lock(arg->mutex); // regiao critica: leitura+escrita da global
    total_geral += arg->soma;
    pthread_mutex_unlock(arg->mutex);
    return NULL;
}

char *read_line(int fd) {
    int cap = 16, len = 0;
    char *buf = malloc(cap);
    char c;
    int lidos;

    while ((lidos = read(fd, &c, 1)) > 0) {
        if (c == '\n') break;
        if (len + 1 >= cap) {         // +1 para o '\0'
            cap *= 2;
            buf = realloc(buf, cap);
        }
        buf[len++] = c;
    }

    if (lidos <= 0 && len == 0) {     // EOF sem nenhum caractere lido
        free(buf);
        return NULL;
    }
    buf[len] = '\0';
    return buf;
}

int main(int argc, char *argv[]) {
    if (argc != 3) {
        printf("USO: ./q4 <max_threads> <arquivo>\n");
        return 1;
    }
    int max_threads = atoi(argv[1]);
    int fd = open(argv[2], O_RDONLY);
    if (fd == -1) {
        perror("open");
        return 1;
    }

    sem_t limite;
    sem_init(&limite, 0, max_threads);     // valor inicial = numero de vagas
    pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;

    int cap = 2, n = 0;
    pthread_t *tids = malloc(sizeof(pthread_t) * cap);
    t_arg **args = malloc(sizeof(t_arg *) * cap);

    char *linha;
    while ((linha = read_line(fd)) != NULL) {
        if (n == cap) {
            cap *= 2;
            tids = realloc(tids, sizeof(pthread_t) * cap);
            args = realloc(args, sizeof(t_arg *) * cap);
        }
        t_arg *a = malloc(sizeof(t_arg));  // cada thread tem SEU argumento no heap
        a->id = n;
        a->linha = linha;
        a->soma = 0;
        a->limite = &limite;
        a->mutex = &mutex;
        args[n] = a;
        pthread_create(&tids[n], NULL, soma_thread, a);
        n++;
    }
    close(fd);

    for (int i = 0; i < n; i++) {
        pthread_join(tids[i], NULL);
        printf("LINHA %02d SOMA %ld\n", args[i]->id, args[i]->soma);
        free(args[i]->linha);
        free(args[i]);
    }
    printf("TOTAL: %ld\n", total_geral);

    free(args);
    free(tids);
    sem_destroy(&limite);
    pthread_mutex_destroy(&mutex);
    return 0;
}

Verificado: time ./q4 2 numeros.txt levou 3,0 s, e o valgrind deu "All heap blocks were freed" e "0 errors".
